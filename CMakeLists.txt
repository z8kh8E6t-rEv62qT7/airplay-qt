cmake_minimum_required(VERSION 3.25)
project(AirPlayQt VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AirPlayQt requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(AIRPLAY_BUILD_STANDALONE "Build the standalone ASIO application" ON)
option(AIRPLAY_BUILD_VST3 "Build the VST3 effect" ON)
option(AIRPLAY_VST_RATE_DIAGNOSTICS "Compile VST input/send rate diagnostics" OFF)
set(ASIO_SDK_ROOT "E:/MSYS/clang64/include/asiosdk" CACHE PATH "Steinberg ASIO SDK root")
if(AIRPLAY_BUILD_STANDALONE AND NOT EXISTS "${ASIO_SDK_ROOT}/common/iasiodrv.h")
  message(FATAL_ERROR "Set ASIO_SDK_ROOT to the SDK containing common/iasiodrv.h")
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Network Test)
find_package(OpenSSL REQUIRED)
if(MSVC)
  # The VST3 SDK removes global /EHsc; our RAII error paths require unwinding.
  add_compile_options(/utf-8 /Zc:__cplusplus /EHsc)
  include(cmake/LibPlistMsvc.cmake)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0>=2.7)
  add_library(LibPlist::LibPlist ALIAS PkgConfig::PLIST)
endif()
add_library(airplay_core STATIC
  src/app/Settings.cpp
  src/audio/PcmConverter.cpp src/app/SessionController.cpp
  src/airplay/Crypto.cpp src/airplay/RtspClient.cpp
  src/airplay/ReceiverEndpoint.cpp src/airplay/ReceiverDiscovery.cpp
  src/airplay/PtpClock.cpp src/airplay/AirPlaySession.cpp)
target_include_directories(airplay_core PUBLIC src)
if(AIRPLAY_VST_RATE_DIAGNOSTICS)
  target_compile_definitions(airplay_core PUBLIC AIRPLAY_VST_RATE_DIAGNOSTICS)
endif()
target_compile_definitions(airplay_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(airplay_core PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto LibPlist::LibPlist ole32 advapi32 winmm dnsapi)
if(MSVC)
  target_compile_options(airplay_core PRIVATE /W4)
else()
  target_compile_options(airplay_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(airplay_ui STATIC src/ui/StreamingPanel.cpp)
target_link_libraries(airplay_ui PUBLIC airplay_core Qt6::Widgets)
if(AIRPLAY_BUILD_STANDALONE)
  add_library(airplay_asio STATIC src/audio/AsioCapture.cpp src/audio/CaptureTiming.cpp)
  target_include_directories(airplay_asio PUBLIC "${ASIO_SDK_ROOT}/common")
  target_link_libraries(airplay_asio PUBLIC airplay_core)
  add_executable(AirPlayQt WIN32 src/app/main.cpp src/app/Controller.cpp src/ui/MainWindow.cpp)
  target_link_libraries(AirPlayQt PRIVATE airplay_asio airplay_ui)
  if(MSVC)
    include(cmake/DeployStandalone.cmake)
  endif()
endif()
include(CTest)
if(AIRPLAY_BUILD_VST3)
  add_subdirectory(src/vst3)
endif()
if(BUILD_TESTING)
  set(test_modules airplay)
  if(AIRPLAY_BUILD_STANDALONE)
    list(APPEND test_modules audio app)
  endif()
  foreach(module IN LISTS test_modules)
    add_executable(test_${module} tests/${module}/Tests.cpp)
    target_link_libraries(test_${module} PRIVATE airplay_core Qt6::Test)
    add_test(NAME ${module} COMMAND test_${module}
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_${module}.txt,txt
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_${module}.xml,junitxml)
  endforeach()
  add_executable(test_discovery tests/airplay/DiscoveryTests.cpp)
  target_link_libraries(test_discovery PRIVATE airplay_core Qt6::Test)
  add_test(NAME discovery COMMAND test_discovery
    -o ${CMAKE_CURRENT_BINARY_DIR}/test_discovery.txt,txt
    -o ${CMAKE_CURRENT_BINARY_DIR}/test_discovery.xml,junitxml)
  if(AIRPLAY_BUILD_STANDALONE)
  target_sources(test_app PRIVATE src/app/Controller.cpp src/ui/MainWindow.cpp)
  target_link_libraries(test_audio PRIVATE airplay_asio)
  target_link_libraries(test_app PRIVATE airplay_asio airplay_ui)
  set_tests_properties(app PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(test_live tests/audio/Live.cpp src/app/Controller.cpp)
  target_link_libraries(test_live PRIVATE airplay_asio Qt6::Widgets)
  add_executable(test_probe tests/audio/Probe.cpp)
  target_link_libraries(test_probe PRIVATE airplay_asio Qt6::Widgets)
  add_executable(test_record tests/audio/Record.cpp)
  target_link_libraries(test_record PRIVATE airplay_asio Qt6::Widgets)
  endif()
endif()
