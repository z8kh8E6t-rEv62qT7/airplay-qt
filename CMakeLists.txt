cmake_minimum_required(VERSION 3.25)
set(AIRPLAY_RELEASE_VERSION "0.1.0" CACHE STRING "Numeric X.Y.Z application and VST3 version")
if(NOT AIRPLAY_RELEASE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "AIRPLAY_RELEASE_VERSION must be X.Y.Z without leading zeroes")
endif()
project(AirPlayQt VERSION ${AIRPLAY_RELEASE_VERSION} LANGUAGES C CXX)
if((NOT WIN32 AND NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AirPlayQt requires 64-bit Windows, macOS or Linux")
endif()
set(AIRPLAY_LINUX OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(AIRPLAY_LINUX ON)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
option(AIRPLAY_BUILD_STANDALONE "Build the standalone audio input application" ON)
option(AIRPLAY_BUILD_CLI "Build the Linux command-line audio input application" OFF)
if(AIRPLAY_BUILD_CLI AND NOT AIRPLAY_LINUX)
  message(FATAL_ERROR "AIRPLAY_BUILD_CLI requires Linux")
endif()
if(AIRPLAY_LINUX)
  option(AIRPLAY_BUILD_VST3 "Build the VST3 effect" OFF)
  if(AIRPLAY_BUILD_VST3)
    message(FATAL_ERROR "Linux VST3 is not supported; use AIRPLAY_BUILD_VST3=OFF")
  endif()
else()
  option(AIRPLAY_BUILD_VST3 "Build the VST3 effect" ON)
endif()
option(AIRPLAY_VST_RATE_DIAGNOSTICS "Compile VST input/send rate diagnostics" OFF)
option(AIRPLAY_CLI_TEST "Compile the macOS standalone CLI acceptance entry point" OFF)
if(AIRPLAY_CLI_TEST AND (NOT APPLE OR NOT AIRPLAY_BUILD_STANDALONE))
  message(FATAL_ERROR "AIRPLAY_CLI_TEST requires the macOS standalone application")
endif()
set(ASIO_SDK_ROOT "E:/MSYS/clang64/include/asiosdk" CACHE PATH "Steinberg ASIO SDK root")
if(WIN32 AND AIRPLAY_BUILD_STANDALONE AND NOT EXISTS "${ASIO_SDK_ROOT}/common/iasiodrv.h")
  message(FATAL_ERROR "Set ASIO_SDK_ROOT to the SDK containing common/iasiodrv.h")
endif()
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network)
if(AIRPLAY_BUILD_STANDALONE OR AIRPLAY_BUILD_VST3)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
endif()
find_package(OpenSSL REQUIRED)
if(AIRPLAY_LINUX)
  find_package(Qt6 6.5 REQUIRED COMPONENTS DBus)
endif()
if(MSVC)
  # The VST3 SDK removes global /EHsc; our RAII error paths require unwinding.
  add_compile_options(/utf-8 /Zc:__cplusplus /EHsc)
  include(cmake/LibPlistMsvc.cmake)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PLIST REQUIRED IMPORTED_TARGET libplist-2.0>=2.7)
  add_library(LibPlist::LibPlist ALIAS PkgConfig::PLIST)
endif()
add_library(airplay_core STATIC
  src/audio/InputCapture.h
  src/app/Settings.cpp src/app/Message.cpp
  src/audio/PcmConverter.cpp src/app/SessionController.cpp
  src/airplay/Crypto.cpp src/airplay/RtspClient.cpp src/airplay/NetworkBinding.cpp
  src/airplay/EventChannel.cpp src/airplay/DacpServer.cpp src/airplay/NowPlaying.cpp
  src/airplay/ServiceAdvertisement.h
  src/airplay/ReceiverEndpoint.cpp src/airplay/ReceiverDiscovery.h
  src/airplay/PtpClock.cpp src/airplay/AirPlaySession.cpp)
target_include_directories(airplay_core PUBLIC src)
target_compile_definitions(airplay_core PUBLIC AIRPLAY_VERSION="${PROJECT_VERSION}")
if(AIRPLAY_VST_RATE_DIAGNOSTICS)
  target_compile_definitions(airplay_core PUBLIC AIRPLAY_VST_RATE_DIAGNOSTICS)
endif()
target_link_libraries(airplay_core PUBLIC Qt6::Core Qt6::Network OpenSSL::Crypto LibPlist::LibPlist)
if(WIN32)
  target_sources(airplay_core PRIVATE src/airplay/ReceiverDiscoveryWindows.cpp src/airplay/ServiceAdvertisementWindows.cpp)
  target_compile_definitions(airplay_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(airplay_core PUBLIC ole32 advapi32 winmm dnsapi ws2_32)
elseif(APPLE)
  target_sources(airplay_core PRIVATE src/airplay/ReceiverDiscoveryMac.cpp src/airplay/ServiceAdvertisementMac.cpp)
  target_link_libraries(airplay_core PRIVATE "-framework CoreFoundation")
else()
  target_sources(airplay_core PRIVATE src/airplay/AvahiObject.cpp
    src/airplay/ReceiverDiscoveryLinux.cpp src/airplay/ServiceAdvertisementLinux.cpp)
  target_link_libraries(airplay_core PUBLIC Qt6::DBus)
endif()
if(MSVC)
  target_compile_options(airplay_core PRIVATE /W4)
else()
  target_compile_options(airplay_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIRPLAY_BUILD_STANDALONE OR AIRPLAY_BUILD_VST3)
  add_library(airplay_ui STATIC src/ui/StreamingPanel.cpp)
  target_link_libraries(airplay_ui PUBLIC airplay_core Qt6::Widgets)
endif()
if(AIRPLAY_BUILD_STANDALONE OR AIRPLAY_BUILD_CLI)
  if(WIN32)
    add_library(airplay_asio STATIC src/audio/AsioCapture.cpp src/audio/CaptureTiming.cpp)
    target_include_directories(airplay_asio PUBLIC "${ASIO_SDK_ROOT}/common")
    target_link_libraries(airplay_asio PUBLIC airplay_core)
    set(capture_target airplay_asio)
  elseif(APPLE)
    add_library(airplay_coreaudio STATIC src/audio/CoreAudioCapture.cpp
      src/audio/CoreAudioCaptureSession.cpp src/audio/CoreAudioLoopbackCapture.mm)
    target_link_libraries(airplay_coreaudio PUBLIC airplay_core PRIVATE "-framework CoreAudio" "-framework Foundation" Qt6::Gui)
    set(capture_target airplay_coreaudio)
  else()
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=1.4)
    add_library(airplay_pipewire STATIC src/audio/PipeWireCapture.cpp
      src/audio/PipeWireCatalog.cpp src/audio/BluezCatalog.cpp src/audio/BluetoothVolume.cpp)
    target_link_libraries(airplay_pipewire PUBLIC airplay_core PRIVATE PkgConfig::PIPEWIRE)
    target_compile_options(airplay_pipewire PRIVATE -Wall -Wextra -Wpedantic)
    set(capture_target airplay_pipewire)
  endif()
  add_library(airplay_controller STATIC src/app/Controller.cpp)
  target_link_libraries(airplay_controller PUBLIC ${capture_target})
endif()
if(AIRPLAY_BUILD_CLI)
  add_executable(AirPlayQtCli src/app/LinuxCli.cpp)
  target_link_libraries(AirPlayQtCli PRIVATE airplay_controller PkgConfig::PIPEWIRE)
  target_compile_options(AirPlayQtCli PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIRPLAY_BUILD_STANDALONE)
  add_executable(AirPlayQt WIN32 src/app/main.cpp src/ui/MainWindow.cpp)
  target_link_libraries(AirPlayQt PRIVATE airplay_controller airplay_ui)
  if(APPLE)
    if(AIRPLAY_CLI_TEST)
      target_sources(AirPlayQt PRIVATE src/app/Cli.cpp)
      target_compile_definitions(AirPlayQt PRIVATE AIRPLAY_CLI_TEST)
    endif()
    set_target_properties(AirPlayQt PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/cmake/MacInfo.plist.in"
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.airplayqt.app"
      MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
    # Import the Darwin permission backend declared by the custom Info.plist.
    qt_finalize_target(AirPlayQt)
  endif()
  if(MSVC)
    include(cmake/DeployStandalone.cmake)
  endif()
  if(AIRPLAY_LINUX)
    include(GNUInstallDirs)
    install(TARGETS AirPlayQt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES doc/linux/airplayqt.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  endif()
endif()
if(AIRPLAY_LINUX AND (AIRPLAY_BUILD_STANDALONE OR AIRPLAY_BUILD_CLI))
  include(GNUInstallDirs)
  if(AIRPLAY_BUILD_CLI)
    install(TARGETS AirPlayQtCli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  endif()
  install(FILES doc/linux/README.md doc/linux/README.en.md doc/linux/51-airplayqt-bluetooth.conf.example
    doc/linux/70-airplayqt-avrcp.rules.example
    DESTINATION ${CMAKE_INSTALL_DOCDIR})
endif()
if(AIRPLAY_BUILD_VST3)
  add_subdirectory(src/vst3)
endif()
if(BUILD_TESTING)
  if(TARGET airplay_ui)
    add_executable(test_common_app tests/app/CommonTests.cpp)
    target_link_libraries(test_common_app PRIVATE airplay_ui Qt6::Test)
    add_test(NAME common_app COMMAND test_common_app)
    set_tests_properties(common_app PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  set(test_modules airplay)
  if(WIN32 AND AIRPLAY_BUILD_STANDALONE)
    list(APPEND test_modules audio)
  endif()
  foreach(module IN LISTS test_modules)
    add_executable(test_${module} tests/${module}/Tests.cpp)
    target_link_libraries(test_${module} PRIVATE airplay_core Qt6::Test)
    add_test(NAME ${module} COMMAND test_${module}
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_${module}.txt,txt
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_${module}.xml,junitxml)
  endforeach()
  if(WIN32)
    set(discovery_tests tests/airplay/DiscoveryTests.cpp)
  elseif(APPLE)
    set(discovery_tests tests/airplay/DiscoveryMacTests.cpp)
  else()
    set(discovery_tests tests/airplay/DiscoveryLinuxTests.cpp)
  endif()
  add_executable(test_discovery ${discovery_tests})
  target_link_libraries(test_discovery PRIVATE airplay_core Qt6::Test)
  if(AIRPLAY_LINUX)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    set(discovery_launcher ${DBUS_RUN_SESSION} --)
  endif()
  add_test(NAME discovery COMMAND ${discovery_launcher} $<TARGET_FILE:test_discovery>
    -o ${CMAKE_CURRENT_BINARY_DIR}/test_discovery.txt,txt
    -o ${CMAKE_CURRENT_BINARY_DIR}/test_discovery.xml,junitxml)
  if(AIRPLAY_BUILD_STANDALONE)
    add_executable(test_app tests/app/Tests.cpp src/ui/MainWindow.cpp)
    target_link_libraries(test_app PRIVATE airplay_controller airplay_ui Qt6::Test)
    add_test(NAME app COMMAND test_app
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_app.txt,txt
      -o ${CMAKE_CURRENT_BINARY_DIR}/test_app.xml,junitxml)
    set_tests_properties(app PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  if(WIN32 AND AIRPLAY_BUILD_STANDALONE)
    target_link_libraries(test_audio PRIVATE airplay_asio)
    add_executable(test_live tests/audio/Live.cpp)
    target_link_libraries(test_live PRIVATE airplay_controller Qt6::Widgets)
    add_executable(test_probe tests/audio/Probe.cpp)
    target_link_libraries(test_probe PRIVATE airplay_asio Qt6::Widgets)
    add_executable(test_record tests/audio/Record.cpp)
    target_link_libraries(test_record PRIVATE airplay_asio Qt6::Widgets)
  endif()
endif()

if(BUILD_TESTING AND AIRPLAY_LINUX AND (AIRPLAY_BUILD_STANDALONE OR AIRPLAY_BUILD_CLI))
  add_executable(test_pipewire tests/audio/PipeWireTests.cpp)
  target_link_libraries(test_pipewire PRIVATE airplay_pipewire PkgConfig::PIPEWIRE Qt6::Test)

  set_tests_properties(bluetooth PROPERTIES TIMEOUT 30)
