if(MSVC)
  target_compile_definitions(airplay_vst3 PRIVATE AIRPLAY_PACKAGED_RUNTIME)
  add_library(AirPlayQtEngine MODULE Factory.cpp)
  smtg_target_add_library_main(AirPlayQtEngine)
  target_link_libraries(AirPlayQtEngine PRIVATE airplay_vst3)
  set(SMTG_RUN_VST_VALIDATOR OFF)
  set(SMTG_CREATE_MODULE_INFO OFF)
  smtg_add_vst3plugin(AirPlayQtVst3 PACKAGE_NAME "AirPlayQt" SOURCES_LIST Loader.cpp)
  get_target_property(loader_sources AirPlayQtVst3 SOURCES)
  list(FILTER loader_sources EXCLUDE REGEX "/source/main/dllmain.cpp$")
  set_property(TARGET AirPlayQtVst3 PROPERTY SOURCES "${loader_sources}")
  set_target_properties(AirPlayQtVst3 PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY MultiThreaded)
  # SDK directory options append /MD independently of the CMake runtime property.
  target_compile_options(AirPlayQtVst3 PRIVATE /MT)
  target_link_libraries(AirPlayQtVst3 PRIVATE bcrypt)
  get_filename_component(deploy_prefix "${OPENSSL_INCLUDE_DIR}" DIRECTORY)
  find_file(AIRPLAY_QWINDOWS qwindows.dll PATHS "${deploy_prefix}/lib/qt6/plugins/platforms" NO_DEFAULT_PATH REQUIRED)
  configure_file("${PROJECT_SOURCE_DIR}/cmake/DeployVstRuntime.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/DeployVstRuntime.cmake" @ONLY)
  add_custom_target(AirPlayQtVst3Runtime ALL
    COMMAND "${CMAKE_COMMAND}" "-DPLUGIN=$<TARGET_FILE:AirPlayQtVst3>"
      "-DENGINE=$<TARGET_FILE:AirPlayQtEngine>" "-DQT_BIN=$<TARGET_FILE_DIR:Qt6::Core>"
      -P "${CMAKE_CURRENT_BINARY_DIR}/DeployVstRuntime.cmake"
    DEPENDS AirPlayQtVst3 AirPlayQtEngine VERBATIM)
  get_target_property(package_path AirPlayQtVst3 SMTG_PLUGIN_PACKAGE_PATH)
  set_property(TARGET AirPlayQtVst3Runtime PROPERTY SMTG_PLUGIN_PACKAGE_PATH "${package_path}")
  set(SMTG_CREATE_MODULE_INFO ON)
  smtg_target_create_module_info_file(AirPlayQtVst3Runtime)
  smtg_target_run_vst_validator(AirPlayQtVst3Runtime)
else()
  smtg_add_vst3plugin(AirPlayQtVst3 PACKAGE_NAME "AirPlayQt" SOURCES_LIST Factory.cpp)
  target_link_libraries(AirPlayQtVst3 PRIVATE airplay_vst3)
endif()
